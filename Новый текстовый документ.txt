    const size = document.getElementById('size');
    const construction = document.getElementById('construction');
    const mattressSelect = document.getElementById('mattressSelect');
    const pillows = document.getElementById('pillows');
    const applyQuiz = document.getElementById('applyQuiz');

    const sumBed = document.getElementById('sumBed');
    const sumConstruction = document.getElementById('sumConstruction');
    const sumMattress = document.getElementById('sumMattress');
    const sumPillows = document.getElementById('sumPillows');
    const sumDiscount = document.getElementById('sumDiscount');
    const totalPrice = document.getElementById('totalPrice');
    const depositText = document.getElementById('depositText');
    const inst3 = document.getElementById('inst3');
    const inst6 = document.getElementById('inst6');
    const inst12 = document.getElementById('inst12');

    let quizDiscount = 0;

    function formatPrice(num){
      if(num === 0) return '0 ₽';
      return new Intl.NumberFormat('ru-RU').format(num) + ' ₽';
    }

    function updateSummary(){
      const bed = Number(size.value);
      const constr = Number(construction.value);
      const mattress = Number(mattressSelect.value);
      const pillowPrice = pillows.checked ? 3000 : 0;

      let total = bed + constr + mattress + pillowPrice - quizDiscount;
      if(total < 0) total = 0;

      sumBed.textContent = bed === 0 ? 'Индивидуально' : formatPrice(bed);
      sumConstruction.textContent = formatPrice(constr);
      sumMattress.textContent = formatPrice(mattress);
      sumPillows.textContent = formatPrice(pillowPrice);
      sumDiscount.textContent = quizDiscount ? '− ' + formatPrice(quizDiscount) : '0 ₽';
      totalPrice.textContent = bed === 0 ? 'По запросу' : formatPrice(total);
      depositText.textContent = bed === 0 ? 'Предоплата 60% рассчитывается после согласования.' : 'Предоплата 60%: ' + formatPrice(Math.round(total * 0.6));
      inst3.textContent = bed === 0 ? '—' : formatPrice(Math.round(total / 3));
      inst6.textContent = bed === 0 ? '—' : formatPrice(Math.round(total / 6));
      inst12.textContent = bed === 0 ? '—' : formatPrice(Math.round(total / 12));
    }

    [size, construction, mattressSelect, pillows].forEach(el => el.addEventListener('change', updateSummary));

    applyQuiz.addEventListener('click', () => {
      quizDiscount = quizDiscount ? 0 : 500;
      applyQuiz.textContent = quizDiscount ? 'Бонус квиза применен ✓' : 'Применить бонус квиза −500 ₽';
      updateSummary();
    });

    document.querySelectorAll('.tab').forEach(tab => {
      tab.addEventListener('click', () => {
        document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
        document.querySelectorAll('.tab-content').forEach(c => c.style.display = 'none');
        tab.classList.add('active');
        document.getElementById(tab.dataset.tab).style.display = 'grid';
      });
    });

    document.querySelectorAll('.mattress').forEach(card => {
      card.addEventListener('click', () => {
        document.querySelectorAll('.mattress').forEach(c => c.classList.remove('active'));
        card.classList.add('active');
        const price = card.dataset.price;
        const name = card.dataset.name;
        [...mattressSelect.options].forEach(opt => {
          if(opt.textContent.includes(name)){
            mattressSelect.value = price;
          }
        });
        updateSummary();
        document.getElementById('configurator').scrollIntoView({behavior:'smooth', block:'start'});
      });
    });

    const observer = new IntersectionObserver((entries)=>{
      entries.forEach(entry=>{
        if(entry.isIntersecting) entry.target.classList.add('visible');
      });
    }, {threshold: .12});

    document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

    updateSummary();
